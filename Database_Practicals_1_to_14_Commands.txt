PROBLEM 1 – CUSTOMER, BRANCH, ACCOUNT, LOAN (MySQL)
============================================================

CREATE TABLE Customer (
  cust_id INT PRIMARY KEY,
  cust_name VARCHAR(40) NOT NULL,
  city VARCHAR(30),
  phone VARCHAR(15)
);

CREATE TABLE Branch (
  branch_id INT PRIMARY KEY,
  branch_name VARCHAR(40) NOT NULL,
  branch_city VARCHAR(30)
);

CREATE TABLE Account (
  acc_no INT PRIMARY KEY,
  cust_id INT,
  branch_id INT,
  balance DECIMAL(12,2),
  FOREIGN KEY (cust_id) REFERENCES Customer(cust_id),
  FOREIGN KEY (branch_id) REFERENCES Branch(branch_id)
);

CREATE TABLE Loan (
  loan_no INT PRIMARY KEY,
  cust_id INT,
  branch_id INT,
  amount DECIMAL(12,2),
  FOREIGN KEY (cust_id) REFERENCES Customer(cust_id),
  FOREIGN KEY (branch_id) REFERENCES Branch(branch_id)
);

SELECT s.name,e.course,e.marks
FROM Students s JOIN Enrolments e ON s.student_id=e.student_id;

SELECT s.name,e.course,e.marks
FROM Students s LEFT JOIN Enrolments e ON s.student_id=e.student_id;

-- 3. All enrolments with corresponding student names
SELECT e.*,s.name
FROM Enrolments e LEFT JOIN Students s ON e.student_id=s.student_id;

-- 4. All possible combinations of students and courses (Cartesian product)
SELECT s.name,e.course FROM Students s CROSS JOIN Enrolments e;

-- 5. Marks greater than average marks
SELECT s.student_id,s.name,e.marks
FROM Students s JOIN Enrolments e ON s.student_id=e.student_id
WHERE e.marks>(SELECT AVG(marks) FROM Enrolments);

-- 6. Students with marks greater than 80
SELECT DISTINCT s.student_id,s.name,e.marks
FROM Students s JOIN Enrolments e ON s.student_id=e.student_id
WHERE e.marks>80;

-- 7. Students enrolled in DBMS
SELECT s.name,e.marks
FROM Students s JOIN Enrolments e ON s.student_id=e.student_id
WHERE e.course='DBMS';

-- 8. Student ID(s) with highest marks
SELECT student_id FROM Enrolments
WHERE marks=(SELECT MAX(marks) FROM Enrolments);

CREATE OR REPLACE PROCEDURE proc_Grade
IS
BEGIN
  DELETE FROM Result;

  INSERT INTO Result(Roll,Name,Class)
  SELECT rollno,name,
    CASE
      WHEN total_marks BETWEEN 990 AND 1500 THEN 'Distinction'
      WHEN total_marks BETWEEN 900 AND 989 THEN 'First Class'
      WHEN total_marks BETWEEN 825 AND 899 THEN 'Higher Second Class'
      ELSE 'No Category'
    END
  FROM Stud_Marks
  WHERE total_marks<=1500;

  COMMIT;
END;

PROBLEM 5 – AREA OF CIRCLE FUNCTION (Oracle PL/SQL)
============================================================

CREATE TABLE areas (
  radius NUMBER,
  area NUMBER
);

CREATE OR REPLACE FUNCTION circle_area(p_radius NUMBER)
RETURN NUMBER
IS
BEGIN
  RETURN 3.14159*p_radius*p_radius;
END;
/

BEGIN
  FOR r IN 5..9 LOOP
    INSERT INTO areas(radius,area)
    VALUES(r,circle_area(r));
  END LOOP;
  COMMIT;
END;
/

SELECT * FROM areas ORDER BY radius;

CREATE OR REPLACE PROCEDURE Display_Employees
IS
  CURSOR c1 IS
    SELECT * FROM Employee WHERE Salary>30000;

  info c1%ROWTYPE;
BEGIN
  OPEN c1;
  LOOP
    FETCH c1 INTO info;
    EXIT WHEN c1%NOTFOUND;

    DBMS_OUTPUT.PUT_LINE('Employee ID: '||info.Emp_ID);
    DBMS_OUTPUT.PUT_LINE('Employee Name: '||info.Emp_Name);
    DBMS_OUTPUT.PUT_LINE('Department: '||info.Department);
    DBMS_OUTPUT.PUT_LINE('Salary: '||info.Salary);
    DBMS_OUTPUT.PUT_LINE('---------------------');
  END LOOP;
  CLOSE c1;
END;

PROBLEM 8 – MYSQL DATABASE CONNECTIVITY USING PYTHON
============================================================

Install the connector (on a computer where installation is permitted):
python -m pip install mysql-connector-python

Python program (change database, username and password to your own):
------------------------------------------------------------
import mysql.connector

con = mysql.connector.connect(
    host="localhost",
    user="root",
    password="your_password",
    database="college"
)
cur = con.cursor()

def add_student():
    sid = int(input("Student ID: "))
    name = input("Name: ")
    city = input("City: ")
    cur.execute(
        "INSERT INTO Students(student_id,name,city) VALUES(%s,%s,%s)",
        (sid,name,city)
    )
    con.commit()
    print("Record added")

def display_students():
    cur.execute("SELECT * FROM Students")
    for row in cur.fetchall():
        print(row)

def update_student():
    sid = int(input("Student ID to update: "))
    city = input("New city: ")
    cur.execute("UPDATE Students SET city=%s WHERE student_id=%s",(city,sid))
    con.commit()
    print("Record updated")

def delete_student():
    sid = int(input("Student ID to delete: "))
    cur.execute("DELETE FROM Students WHERE student_id=%s",(sid,))
    con.commit()
    print("Record deleted")

while True:
    print("1 Add  2 Display  3 Update  4 Delete  5 Exit")
    choice = input("Choice: ")
    if choice == "1":
        add_student()
    elif choice == "2":
        display_students()
    elif choice == "3":
        update_student()
    elif choice == "4":
        delete_student()
    elif choice == "5":
        break
    else:
        print("Invalid choice")

cur.close()
con.close()

PROBLEM 10 – BANK DATABASE (Oracle SQL)
============================================================

CREATE TABLE Branch (
  branch_name VARCHAR2(40) PRIMARY KEY,
  branch_city VARCHAR2(30) NOT NULL,
  assets NUMBER(14,2)
);

CREATE TABLE Customer (
  cust_name VARCHAR2(40) PRIMARY KEY,
  cust_street VARCHAR2(60),
  cust_city VARCHAR2(30)
);

CREATE TABLE Account (
  Acc_no NUMBER PRIMARY KEY,
  branch_name VARCHAR2(40),
  balance NUMBER(14,2),
  FOREIGN KEY(branch_name) REFERENCES Branch(branch_name)
);

CREATE TABLE Depositor (
  cust_name VARCHAR2(40),
  acc_no NUMBER,
  PRIMARY KEY(cust_name,acc_no),
  FOREIGN KEY(cust_name) REFERENCES Customer(cust_name),
  FOREIGN KEY(acc_no) REFERENCES Account(Acc_no)
);

CREATE TABLE Loan (
  loan_no NUMBER PRIMARY KEY,
  branch_name VARCHAR2(40),
  amount NUMBER(14,2),
  FOREIGN KEY(branch_name) REFERENCES Branch(branch_name)
);

CREATE TABLE Borrower (
  cust_name VARCHAR2(40),
  loan_no NUMBER,
  PRIMARY KEY(cust_name,loan_no),
  FOREIGN KEY(cust_name) REFERENCES Customer(cust_name),
  FOREIGN KEY(loan_no) REFERENCES Loan(loan_no)
);

Use the tables and sample records from Problem 1, but create them in Oracle
using NUMBER and VARCHAR2 types if you are running this problem in Oracle.
The following queries are Oracle SQL.

-- 1. Customers in Pune or Mumbai, alphabetical
SELECT * FROM Customer
WHERE city IN ('Pune','Mumbai')
ORDER BY cust_name;

-- 2. Names containing letter a
SELECT cust_name FROM Customer WHERE LOWER(cust_name) LIKE '%a%';

-- 3. Phone is NOT NULL
SELECT * FROM Customer WHERE phone IS NOT NULL;

-- 4. Pune and name starts P, OR Mumbai
SELECT * FROM Customer
WHERE (city='Pune' AND cust_name LIKE 'P%') OR city='Mumbai';

-- 5. Maximum and average loan amount per branch
SELECT b.branch_name,MAX(l.amount) AS maximum_loan,
       AVG(l.amount) AS average_loan
FROM Branch b JOIN Loan l ON b.branch_id=l.branch_id
GROUP BY b.branch_id,b.branch_name;

-- 6. Branches with total loan amount > 100000
SELECT b.branch_name,SUM(l.amount) AS total_loan
FROM Branch b JOIN Loan l ON b.branch_id=l.branch_id
GROUP BY b.branch_id,b.branch_name
HAVING SUM(l.amount)>100000;

-- 7. Display balance plus 500 without changing stored balance
SELECT acc_no,balance,balance+500 AS balance_after_500
FROM Account;

-- 8. Change branch for account 2005 (branch_id 6 must exist)
UPDATE Account SET branch_id=6 WHERE acc_no=2005;

-- 9. Delete customers whose phone is NULL.
-- Delete dependent rows first to satisfy foreign keys.
DELETE FROM Account WHERE cust_id IN
 (SELECT cust_id FROM Customer WHERE phone IS NULL);
DELETE FROM Loan WHERE cust_id IN
 (SELECT cust_id FROM Customer WHERE phone IS NULL);
DELETE FROM Customer WHERE phone IS NULL;

-- 10(a). Customers with both an account and a loan (INTERSECT)
SELECT cust_id FROM Account
INTERSECT
SELECT cust_id FROM Loan;

-- 10(b). Customers with an account but no loan (MINUS)
SELECT cust_id FROM Account
MINUS
SELECT cust_id FROM Loan;

IMPORTANT NOTES / COMMON EXAM POINTS
============================================================

1. Primary key: uniquely identifies each row; cannot be NULL.
2. Foreign key: refers to a primary/unique key in another table.
3. INNER JOIN: only matching rows.
4. LEFT JOIN: all rows from left table, matching rows from right table if any.
5. CROSS JOIN: every possible pair of rows.
6. WHERE filters rows before grouping.
7. HAVING filters groups after GROUP BY.
8. COUNT(*) counts rows; AVG() calculates average; SUM() totals values.
9. IS NULL / IS NOT NULL are used for NULL checks. Do not write = NULL.
10. LIKE 'P%' means starts with P; LIKE '%a%' means contains a.
11. UNION combines results and removes duplicates.
12. INTERSECT returns common rows; MINUS returns rows in first query but not second
    (Oracle syntax).
13. A primary key already creates a unique index in MySQL/Oracle; avoid creating
    a redundant index on the same key.
14. MongoDB: collection is similar to a table; document is similar to a row;
    field is similar to a column.
15. MongoDB operators: $gt, $gte, $lt, $lte, $ne, $or.
16. MongoDB aggregation: $group groups documents; "$field" refers to a field value.
17. Oracle PL/SQL cursor flow: OPEN -> FETCH -> EXIT WHEN %NOTFOUND -> process
    fetched row -> repeat -> CLOSE.
18. In Oracle row triggers, use :OLD for the previous values and :NEW for new
    values. In a BEFORE UPDATE/DELETE trigger, :OLD contains the old row.
19. Oracle procedure declaration uses IS or AS, not DECLARE after IS.
20. In a cursor declaration, info c1%ROWTYPE; is valid and matches the cursor
    result columns.
