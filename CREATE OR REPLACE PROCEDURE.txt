CREATE OR REPLACE PROCEDURE areaFind(X IN NUMBER,Y IN NUMBER)
IS
PI NUMBER:=3.14;
I NUMBER:=X;
J NUMBER:=Y;
RESULT NUMBER;
BEGIN
WHILE I<=J
LOOP
	RESULT:=PI*I*I;
	INSERT INTO AREAS VALUES(I,RESULT);
	I:=I+1;
END LOOP;
END;

CREATE OR REPLACE TRIGGER update_library
BEFORE UPDATE OR DELETE ON Library
FOR EACH ROW
BEGIN
	IF UPDATING THEN
    INSERT INTO Library_Audit
    VALUES (
        :OLD.book_id,
        :OLD.book_name,
        :OLD.author,
        :OLD.category,
        :OLD.date1,
        'UPDATE'
    );
    
    ELSIF DELETING THEN
    INSERT INTO Library_Audit
    VALUES (
        :OLD.book_id,
        :OLD.book_name,
        :OLD.author,
        :OLD.category,
        :OLD.date1,
        'DELETE'
    );
END IF;
END;

CREATE OR REPLACE PROCEDURE proc_Grade IS
CURSOR C1 IS SELECT ROLL_NO,NAME,TOTAL_MARKS from STUDENT_MARKS;
class varchar2(50);
info c1%rowtype;
BEGIN
OPEN C1;
LOOP
FETCH C1 INTO info;
EXIT WHEN C1%NOTFOUND;

IF info.TOTAL_MARKS<=1500 AND info.TOTAL_MARKS>=990 then
	class:='distinction category';
	
elsif info.TOTAL_MARKS<=989 AND info.TOTAL_MARKS>=900 then
	class:='first class';
	
elsif info.TOTAL_MARKS<=899 AND info.TOTAL_MARKS>=825 then
	class:='Higher Second Class';
	
END IF;
INSERT INTO Result values(info.ROLL_NO,info.NAME,class);
END LOOP;
CLOSE C1;




END;
